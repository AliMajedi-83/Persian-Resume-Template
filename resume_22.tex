\newcommand{\skillbar}[1]{%
  \begin{tikzpicture}[baseline=0pt]
    \foreach \i in {1,...,5} {
      \pgfmathsetmacro{\x}{(\i-1)*0.44}
      \ifnum\i>#1\relax
        \fill[headbg] (\x+0.01, 0.01) rectangle (\x+0.43, 0.19);
      \else
        \fill[primary] (\x+0.01, 0.01) rectangle (\x+0.43, 0.19);
      \fi
    }
    \draw[primary, line width=1pt] (0, 0) rectangle (2.2, 0.2);
  \end{tikzpicture}%
}
